/** * GitHubExport.gs * Compila os dados da planilha em um único JSON e publica no repositório * GitHub (para consumo pelo index.html estático via GitHub Pages). * * Configuração necessária (Editor do Apps Script → Configurações do projeto * → Propriedades do script): * GITHUB_TOKEN - token fine-grained com permissão "Contents: Read and write" * restrito ao repositório abaixo * GITHUB_OWNER - ex.: mauriciooura-a11yke * GITHUB_REPO - ex.: gbr-adm-gru * GITHUB_BRANCH - opcional, padrão "main" * GITHUB_PATH - opcional, padrão "data.json" */ function compilarJSON_() { var eixosPdi = sheetToObjects_('eixos_pdi').map(limparRow_).filter(function (o) { return o.Ativo !== false; }); var mvv = sheetToObjects_('mvv').map(limparRow_).filter(function (o) { return o.Ativo !== false; }); var swot = sheetToObjects_('swot').map(limparRow_).filter(function (o) { return o.Ativo !== false; }); var objetivosRaw = sheetToObjects_('objetivos').map(limparRow_).filter(function (o) { return o.Ativo !== false; }); var riscos = sheetToObjects_('riscos').map(limparRow_).filter(function (o) { return o.Ativo !== false; }); var riscosOE = sheetToObjects_('riscos_oe').map(limparRow_); var swotOE = sheetToObjects_('swot_oe').map(limparRow_); var mvvOE = sheetToObjects_('mvv_oe').map(limparRow_); var mvvEixo = sheetToObjects_('mvv_eixo_pdi').map(limparRow_); var planoAcao = sheetToObjects_('plano_acao').map(limparRow_).filter(function (o) { return o.Ativo !== false; }); var historico = sheetToObjects_('historico_acao').map(limparRow_).filter(function (o) { return o.Ativo !== false; }); // Resolve os eixos do PDI relacionados a cada item de MVV (N:N) mvv.forEach(function (m) { var idsEixo = mvvEixo.filter(function (v) { return v.ID_MVV === m.ID; }).map(function (v) { return v.ID_EixoPDI; }); m.eixosPdiRelacionados = eixosPdi.filter(function (e) { return idsEixo.indexOf(e.ID) !== -1; }); }); // Anexa a cada risco a lista de OEs vinculados (N:N) riscos.forEach(function (r) { r.OEs = riscosOE .filter(function (v) { return v.ID_Risco === r.ID_Risco; }) .map(function (v) { return v.ID_OE; }); }); // Anexa a cada item SWOT a lista de OEs relacionados (N:N) swot.forEach(function (s) { s.OEs = swotOE .filter(function (v) { return v.ID_SWOT === s.ID; }) .map(function (v) { return v.ID_OE; }); }); // Calcula agregados de cada OE a partir dos riscos vinculados, e anexa o SWOT e o MVV relacionados var objetivos = objetivosRaw.map(function (oe) { var agr = calcularAgregadosOE_(oe.ID_OE, riscos, riscosOE); var idsSwot = swotOE.filter(function (v) { return v.ID_OE === oe.ID_OE; }).map(function (v) { return v.ID_SWOT; }); var swotRelacionado = swot.filter(function (s) { return idsSwot.indexOf(s.ID) !== -1; }); var idsMvv = mvvOE.filter(function (v) { return v.ID_OE === oe.ID_OE; }).map(function (v) { return v.ID_MVV; }); var mvvRelacionado = mvv.filter(function (m) { return idsMvv.indexOf(m.ID) !== -1; }); return Object.assign({}, oe, agr, { swotRelacionado: swotRelacionado, mvvRelacionado: mvvRelacionado }); }); return { atualizadoEm: new Date().toISOString(), eixosPdi: eixosPdi, mvv: mvv, swot: swot, objetivos: objetivos, riscos: riscos, planoAcao: planoAcao, historico: historico }; } function api_publicarGitHub() { checkAccess_(); return publicarGitHub_(); } function publicarGitHubManual() { try { var res = publicarGitHub_(); SpreadsheetApp.getUi().alert('Publicado com sucesso em ' + res.atualizadoEm); } catch (e) { SpreadsheetApp.getUi().alert('Erro ao publicar: ' + e.message); } } function publicarGitHub_() { var props = PropertiesService.getScriptProperties(); var token = props.getProperty('GITHUB_TOKEN'); var owner = props.getProperty('GITHUB_OWNER'); var repo = props.getProperty('GITHUB_REPO'); var branch = props.getProperty('GITHUB_BRANCH') || 'main'; var path = props.getProperty('GITHUB_PATH') || 'data.json'; if (!token || !owner || !repo) { throw new Error('Configure GITHUB_TOKEN, GITHUB_OWNER e GITHUB_REPO nas Propriedades do Script antes de publicar.'); } var json = compilarJSON_(); var conteudo = JSON.stringify(json, null, 2); var conteudoB64 = Utilities.base64Encode(Utilities.newBlob(conteudo).getBytes()); var baseUrl = 'https://api.github.com/repos/' + owner + '/' + repo + '/contents/' + path; // 1) Obtém o sha do arquivo atual, se existir (necessário para atualização) var sha = null; var getResp = UrlFetchApp.fetch(baseUrl + '?ref=' + branch, { method: 'get', headers: { Authorization: 'Bearer ' + token, Accept: 'application/vnd.github+json' }, muteHttpExceptions: true }); if (getResp.getResponseCode() === 200) { sha = JSON.parse(getResp.getContentText()).sha; } var payload = { message: 'Atualização automática de dados — GBR ' + Utilities.formatDate(new Date(), 'America/Sao_Paulo', 'dd/MM/yyyy HH:mm'), content: conteudoB64, branch: branch }; if (sha) payload.sha = sha; var putResp = UrlFetchApp.fetch(baseUrl, { method: 'put', contentType: 'application/json', headers: { Authorization: 'Bearer ' + token, Accept: 'application/vnd.github+json' }, payload: JSON.stringify(payload), muteHttpExceptions: true }); var codigo = putResp.getResponseCode(); if (codigo !== 200 && codigo !== 201) { throw new Error('Falha ao publicar no GitHub (HTTP ' + codigo + '): ' + putResp.getContentText()); } return { ok: true, atualizadoEm: json.atualizadoEm }; }
